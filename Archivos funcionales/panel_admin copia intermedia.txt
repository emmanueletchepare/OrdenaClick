function calcularTotalRegistro(){

    const neto =
        parseFloat(document.getElementById("neto").value) || 0;

    const exento =
        parseFloat(document.getElementById("exento").value) || 0;

    const iva21 =
        parseFloat(document.getElementById("iva21").value) || 0;

    const iva27 =
        parseFloat(document.getElementById("iva27").value) || 0;

    const iva105 =
        parseFloat(document.getElementById("iva105").value) || 0;

    const percepciones =
        parseFloat(document.getElementById("percepciones").value) || 0;

    const ajuste =
        parseFloat(document.getElementById("ajuste").value) || 0;

    const total =
        neto +
        exento +
        iva21 +
        iva27 +
        iva105 +
        percepciones +
        ajuste;

    document.getElementById("total_registro").value =
        total.toFixed(2);
}

function calcularIVA(porcentaje,idDestino){

    const neto =
        parseFloat(document.getElementById("neto").value) || 0;

    const iva =
        neto * porcentaje;

    document.getElementById(idDestino).value =
        iva.toFixed(2);

    calcularTotalRegistro();
}

function togglePago(){

    const bloque =
        document.getElementById("bloquePago");

    const boton =
        document.getElementById("btnPago");

    if(bloque.style.display==="none"){

        bloque.style.display="block";

        boton.innerHTML="▼ Datos del Pago";

    }else{

        bloque.style.display="none";

        boton.innerHTML="► Datos del Pago";

    }

}

function toggleCheque(){
